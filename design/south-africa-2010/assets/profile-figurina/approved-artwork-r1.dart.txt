// Normalized 160x70 detail, independent of the card's content and height.
class ProfileCornerSweep extends CustomPainter {
  const ProfileCornerSweep();

  @override
  void paint(Canvas canvas, Size size) {
    canvas.save();
    canvas.scale(size.width / 160, size.height / 70);
    final yellow = Path()
      ..moveTo(0, 70)
      ..cubicTo(64, 56, 116, 32, 160, 0)
      ..lineTo(160, 17)
      ..cubicTo(104, 49, 49, 64, 0, 70)
      ..close();
    final green = Path()
      ..moveTo(27, 70)
      ..cubicTo(83, 57, 126, 38, 160, 23)
      ..lineTo(160, 36)
      ..cubicTo(112, 58, 66, 67, 27, 70)
      ..close();
    final blue = Path()
      ..moveTo(64, 70)
      ..cubicTo(106, 59, 137, 49, 160, 41)
      ..lineTo(160, 53)
      ..cubicTo(123, 67, 101, 70, 64, 70)
      ..close();
    final red = Path()
      ..moveTo(112, 70)
      ..cubicTo(135, 65, 148, 61, 160, 58)
      ..lineTo(160, 70)
      ..close();
    _fill(canvas, yellow, AppColors.yellow, light: .06, dark: .04);
    _fill(canvas, green, AppColors.success, light: .06, dark: .08);
    _fill(canvas, blue, AppColors.primary, light: .06, dark: .08);
    _fill(canvas, red, AppColors.error, light: .06, dark: .08);
    canvas.restore();
  }

  @override
  bool shouldRepaint(covariant ProfileCornerSweep oldDelegate) => false;
}

void _fill(
  Canvas canvas,
  Path path,
  Color color, {
  double light = .12,
  double dark = .12,
}) {
  canvas.drawPath(
    path,
    Paint()
      ..shader = LinearGradient(
        begin: Alignment.topLeft,
        end: Alignment.bottomRight,
        colors: [
          Color.lerp(color, AppColors.surface, light)!,
          color,
          Color.lerp(color, AppColors.ink, dark)!,
        ],
        stops: const [0, .52, 1],
      ).createShader(path.getBounds()),
  );
}

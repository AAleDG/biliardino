class _FieldPainter extends CustomPainter {
  const _FieldPainter();
  @override
  void paint(Canvas canvas, Size size) {
    canvas.save(); canvas.scale(size.width / 440, size.height / 310);
    final silhouette = Path()..moveTo(0, 64)..cubicTo(95, 110, 148, 74, 218, 94)
      ..cubicTo(305, 75, 361, 108, 440, 60)..lineTo(440, 238)
      ..cubicTo(403, 295, 348, 310, 246, 310)..lineTo(194, 310)
      ..cubicTo(100, 310, 35, 289, 0, 238)..close();
    canvas.save(); canvas.clipPath(silhouette);
    canvas.drawRect(const Rect.fromLTWH(0, 0, 440, 310), Paint()..color=AppColors.teamOne);
    canvas.drawPath(Path()..moveTo(220, 0)..lineTo(440,0)..lineTo(440,310)..lineTo(225,310)
      ..cubicTo(243,237,238,167,220,110)..close(), Paint()..color=AppColors.teamTwo);
    canvas.drawPath(Path()..moveTo(204,78)..lineTo(230,78)..cubicTo(243,157,259,227,232,310)
      ..lineTo(196,310)..cubicTo(201,220,187,141,204,78)..close(), Paint()..color=AppColors.ink);
    final white=Paint()..color=AppColors.background..style=PaintingStyle.stroke..strokeWidth=5;
    canvas.drawPath(Path()..moveTo(-10,229)..cubicTo(32,278,68,295,113,308),white);
    canvas.drawPath(Path()..moveTo(450,232)..cubicTo(410,279,375,297,330,309),white);
    canvas.restore(); canvas.restore();
  }
  @override
  bool shouldRepaint(_FieldPainter oldDelegate) => false;
}

class _EdgeRibbonPainter extends CustomPainter {
  const _EdgeRibbonPainter();
  @override
  void paint(Canvas canvas, Size size) {
    canvas.save(); canvas.scale(size.width / 440, 1);
    void fill(Path path, Color color) => canvas.drawPath(path, Paint()..color = color);
    fill(Path()..moveTo(0,0)..lineTo(190,0)..cubicTo(125,47,47,103,0,179)..close(), AppColors.yellow);
    fill(Path()..moveTo(242,0)..lineTo(440,0)..lineTo(440,171)
      ..cubicTo(411,111,338,65,242,0)..close(), AppColors.yellow);
    fill(Path()..moveTo(0,160)..cubicTo(30,118,58,85,89,63)
      ..cubicTo(51,121,30,157,23,182)..lineTo(0,189)..close(), AppColors.primary);
    fill(Path()..moveTo(440,192)..cubicTo(421,159,411,140,384,117)
      ..cubicTo(421,138,434,157,440,166)..close(), AppColors.success);
    final h=size.height;
    fill(Path()..moveTo(0,h-175)..cubicTo(31,h-94,71,h-37,184,h)
      ..lineTo(117,h)..cubicTo(38,h-43,18,h-88,0,h-114)..close(), AppColors.success);
    fill(Path()..moveTo(440,h-185)..lineTo(440,h-125)
      ..cubicTo(409,h-70,370,h-39,324,h-14)..lineTo(293,h-12)
      ..cubicTo(360,h-63,398,h-109,440,h-185)..close(), AppColors.yellow);
    fill(Path()..moveTo(0,h-54)..cubicTo(133,h+3,300,h-10,440,h-40)
      ..lineTo(440,h-20)..cubicTo(313,h+11,178,h+24,0,h-20)..close(), AppColors.primary);
    fill(Path()..moveTo(287,h-3)..cubicTo(348,h-10,400,h-23,440,h-52)
      ..lineTo(440,h-36)..cubicTo(384,h-11,334,h-4,287,h-3)..close(), AppColors.teamTwo);
    canvas.restore();
  }
  @override
  bool shouldRepaint(_EdgeRibbonPainter oldDelegate) => false;
}
